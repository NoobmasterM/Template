vector<int>sieve;
vector<int>divisors;
vector<int>sumdiv;
vector<int>totient;
vector<int>spf;


void initPrimes(int n){
    sieve.assign(n+1,0);
    divisors.assign(n+1,0);
    sumdiv.assign(n+1,0);
    totient.assign(n+1,0);
    spf.assign(n+1,1);

   for(int x=2; x<=n; x++){
      if(sieve[x]) continue;
      for(int u=x; u<=n; u+=x) sieve[u]=x;
    }

    for(int i=1; i<=n; i++){
      for(int j=i; j<=n; j+=i) divisors[j]++;
    }
 
    for(int i=1; i<=n; i++){
      for(int j=i; j<=n; j+=i) sumdiv[j]+=i;
    } 

    for(int i=1; i<=n; i++) totient[i]=i;

    for(int i=2; i<=n; i++){
      if(totient[i]!=i) continue;
      for(int j=i; j<=n; j+=i) totient[j]-=totient[j]/i;
    }

    spf[0]=0;
    for(int i=2; i<=n; i++){
      if(spf[i]!=1)continue;
      for(int j=i; j<=n; j+=i) if(spf[j]==1) spf[j]=i;
    }
}

vector<int>primeFactors(int n){
    vector<int>ans;
    while(n!=1){
      ans.push_back(spf[n]);
      n=n/spf[n];
    }
    return ans;
}


vector<long long>primes;

bool check(int n,int pos){return (bool)(n & (1<<pos));}
int setb(int n,int pos){return n=n | (1<<pos);}

void bitwiseSieve(int n){
    vector<int>status(n/32+2,0);
    int i,j;
    int sqrtN=int(sqrt(n));
    for(i=3; i<=sqrtN; i+=2){
        if(check(status[i>>5],i&31)==0){
           for(j=i*i; j<=n; j+=(i<<1)) status[j>>5]=setb(status[j>>5],j&31);
        }
    }
    
    primes.push_back(2);
    for(i=3; i<=n; i+=2){
        if(check(status[i>>5],i&31)==0) primes.push_back(i);
    }
}
