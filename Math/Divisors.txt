long long numberOfDivisors(long long num){
    long long total=1;
    for(long long i=2; i*i<=num; i++){
        if(num%i==0){
            int e=0;
            while(num%i==0){
                e++;
                num/=i;
            }
            total*=(e+1);
        }
    }

    if(num>1) total*=2;
    return total;
}

long long sumOfDivisors(long long num){
    long long total=1;
    for(long long i=2; i*i<=num; i++){
        if(num%i==0){
            int e=0;
            while(num%i==0){
                e++;
                num/=i;
            }

            long long sum=0,pow=1;
            for(int j=e; j>=0; j--){
                sum+=pow;
                pow*=i;
            }
            total*=sum;
        }
    }
    if(num>1) total*=(num+1);
    return total;
}
