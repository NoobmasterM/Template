struct item{long long sum = 0;};

struct segtree{
    long long size;
    vector<item> values;
    vector<long long> lazy;
    item NEUTRAL_ELEMENT = {0};
    item merge(item a, item b){return {a.sum + b.sum};}
    item single(long long v){return {v};}

    void init(long long n){
        size = 1;
        while(size < n) size *= 2;
        values.assign(2 * size, NEUTRAL_ELEMENT);
        lazy.assign(2 * size, 0);
    }

    void build(vector<long long>& a, long long x, long long lx, long long rx){
        if(rx - lx == 1){ if(lx < (long long)a.size()) values[x] = single(a[lx]); return;}
        long long m = (lx + rx) / 2;
        build(a, 2*x+1, lx, m);
        build(a, 2*x+2, m, rx);
        values[x] = merge(values[2*x+1], values[2*x+2]);
    }

    void build(vector<long long>& a){build(a, 0, 0, size);}
    void apply(long long x, long long lx, long long rx, long long v){values[x].sum += v * (rx - lx); lazy[x] += v;}
    void propagate(long long x, long long lx, long long rx){
        if(rx - lx == 1) return;
        if(lazy[x] == 0) return;
        long long m = (lx + rx) / 2;
        apply(2*x+1, lx, m, lazy[x]);
        apply(2*x+2, m, rx, lazy[x]);
        lazy[x] = 0;
    }

    void update(long long l, long long r, long long v, long long x, long long lx, long long rx){
        if(lx >= r || rx <= l) return;
        if(l <= lx && rx <= r){apply(x, lx, rx, v); return;}
        propagate(x, lx, rx);
        long long m = (lx + rx) / 2;
        update(l, r, v, 2*x+1, lx, m);
        update(l, r, v, 2*x+2, m, rx);
        values[x] = merge(values[2*x+1], values[2*x+2]);
    }

    void update(long long l, long long r, long long v){update(l, r, v, 0, 0, size);}

    item calc(long long l, long long r, long long x, long long lx, long long rx){
        if(lx >= r || rx <= l) return NEUTRAL_ELEMENT;
        if(l <= lx && rx <= r) return values[x];
        propagate(x, lx, rx);
        long long m = (lx + rx) / 2;
        item left = calc(l, r, 2*x+1, lx, m);
        item right = calc(l, r, 2*x+2, m, rx);
        return merge(left, right);
    }

    item calc(long long l, long long r){return calc(l, r, 0, 0, size);}
};
