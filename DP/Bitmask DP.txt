 int n,m;
 cin>>n>>m; 

 long long mod=1e9+7;

 vector<vector<long long>>dp(1<<n,vector<long long>(n));
 vector<vector<int>>cadj(n);

 while(m--){
    int u,v;
    cin>>u>>v;
    u--;
    v--;
    cadj[v].push_back(u);
 }

 dp[1][0]=1;
 
 for(int s=2; s<(1<<n); s++){
    if((s&1)==0) continue;

    if((s&(1<<n-1)) && s!=((1<<n)-1)) continue;
 
    for(int d=0; d<n; d++){
        if((s&(1<<d))==0) continue;

        int prev=s^(1<<d);
        for(int j: cadj[d]){
            if(s&(1<<j)){
            dp[s][d]+=dp[prev][j];
            dp[s][d]%=mod;
          }
        }
    }
 }

 cout<<dp[(1<<n)-1][n-1]<<"\n";

-------------------------Shortest HP------------------------
int n,m;
 cin>>n>>m; 

 long long mod=1e9+7;

 vector<vector<long long>>dp(1<<n,vector<long long>(n,1e18));
 vector<vector<int>>cadj(n);

 vector<vector<long long>>ache(n,vector<long long>(n,1e18));

 while(m--){
    long long u,v,w;
    cin>>u>>v>>w;
    u--;
    v--;
    cadj[v].push_back(u);
    ache[u][v]=w;
 }

  for(unsigned int s=1; s<(1<<n); s++){
    if(popcount(s)==1){
      for(int j=0; j<n; j++){
         if((s&(1<<j))) dp[s][j]=0;
      }
    }
  }

 for(unsigned int s=1; s<(1<<n); s++){
    if(popcount(s)<=1) continue;

    for(int d=0; d<n; d++){
        if((s&(1<<d))==0) continue;
        int prev=s^(1<<d);
        for(int j: cadj[d]){
            if(s&(1<<j)){
            dp[s][d]=min(dp[s][d],dp[prev][j]+ache[j][d]);
          }
        }
    }
 }

 long long ans=1e18;

 for(int i=0; i<n; i++) ans=min(ans,dp[(1<<n)-1][i]);
 cout<<ans;

 vector<int>adj(n),check((1<<n),0);
 
 while(m--){
  int u,v;
  cin>>u>>v;
  u--;
  v--;
  adj[u]|=(1<<v);
  adj[v]|=(1<<u);
 }

 for(int i=0; i<(1<<n); i++){
   bool temp=1;
   for(int j=0; j<n; j++){
     if((i&(1<<j))){
        if(((adj[j]|(1<<j))&i)!=i) temp=0;
     }
     check[i]=temp;
   }
 }
